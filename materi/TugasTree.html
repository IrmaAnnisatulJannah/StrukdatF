import networkx as nx
import matplotlib.pyplot as plt

class BinaryTree:
    def __init__(self, root):
        self.key = root
        self.leftChild = None
        self.rightChild = None

    def insertLeft(self, newNode):
        self.leftChild = BinaryTree(newNode)

    def insertRight(self, newNode):
        self.rightChild = BinaryTree(newNode)

    def getLeftChild(self):
        return self.leftChild

    def getRightChild(self):
        return self.rightChild

    def getRootVal(self):
        return self.key


# Membuat Tree
root = BinaryTree("*")

root.insertLeft("+")
root.insertRight("-")

root.getLeftChild().insertLeft("7")
root.getLeftChild().insertRight("3")

root.getRightChild().insertLeft("5")
root.getRightChild().insertRight("2")


# Membuat Grafik Tree
def buatGraf(node, G, parent=None):

    if node is not None:

        node_id = str(id(node))

        G.add_node(node_id,
                   label=node.getRootVal())

        if parent:
            G.add_edge(parent, node_id)

        buatGraf(node.getLeftChild(), G, node_id)
        buatGraf(node.getRightChild(), G, node_id)


G = nx.DiGraph()

buatGraf(root, G)

labels = nx.get_node_attributes(G, 'label')

plt.figure(figsize=(8,6))

pos = nx.spring_layout(G, seed=20)

    G,
    pos,
    labels=labels,
    with_labels=True,
    node_size=2500,
    node_color="lightblue",
    font_size=12
)

plt.title("TREE")
plt.show()

%matplotlib inline
import networkx as nx
import matplotlib.pyplot as plt

# STACK
def stack():
    return []

def push(s, data):
    s.append(data)

def pop(s):
    return s.pop()


# BINARY TREE
class BinaryTree:

    def __init__(self, root):
        self.key = root
        self.leftChild = None
        self.rightChild = None

    def insertLeft(self, value):
        self.leftChild = BinaryTree(value)

    def insertRight(self, value):
        self.rightChild = BinaryTree(value)

    def getLeftChild(self):
        return self.leftChild

    def getRightChild(self):
        return self.rightChild

    def setRootVal(self, obj):
        self.key = obj

    def getRootVal(self):
        return self.key


# PARSE TREE
def buildParseTree(mathExp):

    tokenList = mathExp.split()

    parentStack = stack()

    expTree = BinaryTree('')

    push(parentStack, expTree)

    currentTree = expTree

    print("TRACE PARSE TREE")
    print("-" * 40)

    for token in tokenList:

        current = currentTree.getRootVal()

        if current == '':
            current = "kosong"

        print("Token Dibaca :", token)
        print("Current Node :", current)
        print()

        if token == '(':

            currentTree.insertLeft('')

            push(parentStack, currentTree)

            currentTree = currentTree.getLeftChild()

        elif token not in ['+','-','*','/',')']:

            currentTree.setRootVal(int(token))

            currentTree = pop(parentStack)

        elif token in ['+','-','*','/']:

            currentTree.setRootVal(token)

            currentTree.insertRight('')

            push(parentStack, currentTree)

            currentTree = currentTree.getRightChild()

        elif token == ')':

            currentTree = pop(parentStack)

    return expTree


# Membuat Graph Parse Tree
def buatGraf(node, G, parent=None):

    if node is not None:

        node_id = str(id(node))

        G.add_node(node_id,
                   label=node.getRootVal())

        if parent:
            G.add_edge(parent, node_id)

        buatGraf(node.getLeftChild(), G, node_id)
        buatGraf(node.getRightChild(), G, node_id)


# Jalankan Parse Tree
pt = buildParseTree("( ( 2 + 5 ) * 3 )")

# Tampilkan Parse Tree
G = nx.DiGraph()

buatGraf(pt, G)

labels = nx.get_node_attributes(G, 'label')

plt.figure(figsize=(8,6))

pos = nx.spring_layout(G, seed=15)

nx.draw(
    G,
    pos,
    labels=labels,
    with_labels=True,
    node_size=2500,
    node_color="lightgreen",
    font_size=12
)

plt.title("PARSE TREE")
plt.show()
